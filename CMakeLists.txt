cmake_minimum_required(VERSION 3.21)
project(VideoPlayer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets)
add_library(player_ui STATIC src/playerwindow.cpp src/playerwindow.h src/seekslider.h)
target_include_directories(player_ui PUBLIC src)
target_link_libraries(player_ui PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets)
qt_add_executable(VideoPlayer WIN32 src/main.cpp)
target_link_libraries(VideoPlayer PRIVATE player_ui)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(player_tests tests/player_tests.cpp)
    target_link_libraries(player_tests PRIVATE player_ui Qt6::Test)
    add_test(NAME player_tests COMMAND player_tests)
    set_tests_properties(player_tests PROPERTIES ENVIRONMENT "QT_MEDIA_BACKEND=ffmpeg;QT_QPA_PLATFORM=offscreen")
endif()
