cmake_minimum_required(VERSION 3.21)
project(VideoPlayer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(cmake/ReleaseOptimizations.cmake)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia MultimediaWidgets Network)
option(VIDEO_PLAYER_REQUIRE_QT_LTO "Reject a Qt kit not built with LTO" OFF)
if(VIDEO_PLAYER_REQUIRE_QT_LTO)
    get_filename_component(qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
    if(NOT EXISTS "${qt_prefix}/lto-build.json")
        message(FATAL_ERROR "This build requires the verified LTO Qt kit from scripts/build-qt-lto.py.")
    endif()
    file(READ "${qt_prefix}/lto-build.json" qt_lto_manifest)
    foreach(module qtbase qtshadertools qtsvg qtmultimedia)
        string(JSON module_lto GET "${qt_lto_manifest}" modules ${module} lto)
        if(NOT module_lto)
            message(FATAL_ERROR "Qt module ${module} was not built with LTO")
        endif()
    endforeach()
endif()
add_library(player_ui STATIC src/playerwindow.cpp src/playerwindow.h src/seekslider.h src/singleinstance.cpp src/singleinstance.h)
target_include_directories(player_ui PUBLIC src)
target_link_libraries(player_ui PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::MultimediaWidgets Qt6::Network)
if(WIN32)
    target_link_libraries(player_ui PUBLIC user32 gdi32)
endif()
qt_add_executable(VideoPlayer WIN32 MACOSX_BUNDLE src/main.cpp)
qt_add_resources(VideoPlayer app_icons PREFIX "/" FILES assets/app-icon.png)
if(WIN32)
    target_sources(VideoPlayer PRIVATE assets/app-icon.rc)
    target_include_directories(VideoPlayer PRIVATE assets)
endif()
target_link_libraries(VideoPlayer PRIVATE player_ui)
if(MSVC)
    target_compile_options(player_ui PUBLIC /utf-8)
endif()
if(APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app-icon.icns")
    target_sources(VideoPlayer PRIVATE assets/app-icon.icns)
    set_source_files_properties(assets/app-icon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(VideoPlayer PROPERTIES MACOSX_BUNDLE_ICON_FILE app-icon.icns)
endif()

option(VIDEO_PLAYER_DEPLOY "Install the application with its Qt and FFmpeg runtime" OFF)
if(VIDEO_PLAYER_DEPLOY)
    include(GNUInstallDirs)
    # Keep the AppImage launcher and runtime directory consistent across Linux distros.
    set(CMAKE_INSTALL_LIBDIR lib)
    install(TARGETS VideoPlayer BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES README.md DESTINATION .)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set_target_properties(VideoPlayer PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
        install(FILES packaging/video-player.desktop DESTINATION share/applications)
        install(FILES assets/app-icon.png DESTINATION share/icons/hicolor/512x512/apps RENAME video-player.png)
        # Include FFmpeg and the X11/GL client libraries even if found in system paths.
        # Never bundle glibc or a loader alongside the host's loader.
        set(deploy_options POST_INCLUDE_REGEXES "/lib.*[.]so[.]"
            PRE_EXCLUDE_REGEXES "^ld-linux.*[.]so[.]")
        foreach(library IN ITEMS c m mvec pthread dl rt resolv util anl BrokenLocale thread_db)
            list(APPEND deploy_options "^lib${library}[.]so[.]")
        endforeach()
        list(APPEND deploy_options "^libnss_.*[.]so[.]")
    endif()
    qt_generate_deploy_app_script(TARGET VideoPlayer OUTPUT_SCRIPT deploy_script NO_TRANSLATIONS ${deploy_options})
    install(SCRIPT ${deploy_script})
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(player_tests tests/player_tests.cpp)
    target_link_libraries(player_tests PRIVATE player_ui Qt6::Test)
    add_test(NAME player_tests COMMAND player_tests)
    qt_add_executable(instance_tests tests/instance_tests.cpp)
    target_link_libraries(instance_tests PRIVATE player_ui Qt6::Test)
    add_test(NAME instance_tests COMMAND instance_tests)
    set_tests_properties(player_tests PROPERTIES ENVIRONMENT "QT_MEDIA_BACKEND=ffmpeg;QT_QPA_PLATFORM=offscreen")
endif()
